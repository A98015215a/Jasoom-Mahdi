% Layout shared by the cover page and the title pages.
% \frontPage[text colour command]{margins}{logo}{logo height}{space below logo}{vertical position of the description}{description}
\newcommand{\frontPage}[7][]{%
	\newgeometry{#2}
	\thispagestyle{empty}
	#1

	\begin{center}

	\begin{figure}[h!]
		\begin{center}
		\includegraphics[height=#4]{#3}
		\end{center}
	\end{figure}

	\vspace{#5}

	\Large
	\thesisAuthor

	\vspace{1cm}

	\LARGE
	\textsc{\textbf{\thesisTitle}}

	\Large
	\vspace{-0.1cm}

	\textsc{\thesisSubtitle}

	% The description is as wide as the text area and starts at the left margin
	\begin{textblock*}{\textwidth}(\dimexpr 1in+\oddsidemargin\relax,#6)
	#1
	\normalsize
	\begin{spacing}{1.1}
	\textbf{#7}
	\end{spacing}
	\end{textblock*}

	\large
	\vspace*{\fill}
	#1
	\thesisDate

	\end{center}

	\restoregeometry
	\color{black}
}
